from flask import Flask, render_template_string, request from sentence_transformers import SentenceTransformer import faiss import numpy as np app = Flask(__name__) # Research Papers Database papers = [ { "title": "Smart Farming Using AI", "author": "Dr. Kumar", "abstract": "AI based crop monitoring and yield prediction.", "tags": ["Agriculture", "AI", "IoT"] }, { "title": "Crop Disease Detection", "author": "Dr. Priya", "abstract": "Deep learning for crop disease identification.", "tags": ["Agriculture", "Deep Learning"] }, { "title": "Precision Agriculture", "author": "Dr. Raj", "abstract": "Sensor-based smart farming techniques.", "tags": ["Agriculture", "Sensors"] }, { "title": "Machine Learning for Education", "author": "Dr. Meena", "abstract": "Applying machine learning in education.", "tags": ["AI", "Machine Learning"] } ] # AI Model model = SentenceTransformer("all-MiniLM-L6-v2") # Create Embeddings paper_texts = [ paper["title"] + " " + paper["abstract"] + " " + " ".join(paper["tags"]) for paper in papers ] paper_embeddings = model.encode(paper_texts) # FAISS Index dimension = paper_embeddings.shape[1] index = faiss.IndexFlatL2(dimension) index.add(np.array(paper_embeddings)) HTML = """ ResearchHub
🔬 ResearchHub
Trending Tags

#Agriculture

#AI

#MachineLearning

#IoT

{% if papers %} {% for paper in papers %}

{{paper.title}}

👤 {{paper.author}}

{{paper.abstract}}

{% for tag in paper.tags %} #{{tag}} {% endfor %}
{% endfor %} {% else %}
No papers found.
{% endif %}
Suggested Experts
{% for expert in experts %}

👤 {{expert}}

{% endfor %}
""" @app.route("/", methods=["GET", "POST"]) def home(): query = "" results = papers experts = list(set([p["author"] for p in papers])) if request.method == "POST": query = request.form["search"] query_embedding = model.encode([query]) distances, indices = index.search( np.array(query_embedding), 4 ) results = [] for idx in indices[0]: results.append(papers[idx]) experts = list( set([paper["author"] for paper in results]) ) return render_template_string( HTML, papers=results, experts=experts, query=query ) if __name__ == "__main__": app.run(debug=True)
